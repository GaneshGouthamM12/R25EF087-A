Program:
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#define ROWS 25
#define COLS 60
#define MAX_OBJECTS 50

char canvas[ROWS][COLS];

typedef enum {
    LINE,
    RECTANGLE,
    CIRCLE,
    TRIANGLE
} ShapeType;

typedef struct {
    ShapeType type;
    int p[6];
} Shape;

Shape objects[MAX_OBJECTS];
int objectCount = 0;

/* Function Prototypes */
void clearCanvas();
void displayCanvas();
void plot(int x, int y);
void drawLine(int x1, int y1, int x2, int y2);
void drawRectangle(int x, int y, int width, int height);
void drawCircle(int xc, int yc, int r);
void drawTriangle(int x1, int y1, int x2, int y2, int x3, int y3);
void redrawCanvas();

void addObject(Shape s);
void deleteObject(int index);
void modifyObject(int index, Shape s);

/* Clear Canvas */
void clearCanvas() {
    int i, j;
    for(i = 0; i < ROWS; i++) {
        for(j = 0; j < COLS; j++) {
            canvas[i][j] = '_';
        }
    }
}

/* Display Canvas */
void displayCanvas() {
    int i, j;
    for(i = 0; i < ROWS; i++) {
        for(j = 0; j < COLS; j++) {
            printf("%c", canvas[i][j]);
        }
        printf("\n");
    }
}

/* Plot a Point */
void plot(int x, int y) {
    if(x >= 0 && x < COLS && y >= 0 && y < ROWS)
        canvas[y][x] = '*';
}

/* Draw Line (Bresenham Algorithm) */
void drawLine(int x1, int y1, int x2, int y2) {
    int dx = abs(x2 - x1);
    int dy = abs(y2 - y1);

    int sx = (x1 < x2) ? 1 : -1;
    int sy = (y1 < y2) ? 1 : -1;

    int err = dx - dy;

    while(1) {
        plot(x1, y1);

        if(x1 == x2 && y1 == y2)
            break;

        int e2 = 2 * err;

        if(e2 > -dy) {
            err -= dy;
            x1 += sx;
        }

        if(e2 < dx) {
            err += dx;
            y1 += sy;
        }
    }
}

/* Draw Rectangle */
void drawRectangle(int x, int y, int width, int height) {
    drawLine(x, y, x + width, y);
    drawLine(x, y, x, y + height);
    drawLine(x + width, y, x + width, y + height);
    drawLine(x, y + height, x + width, y + height);
}

/* Draw Circle */
void drawCircle(int xc, int yc, int r) {
    int angle;
    for(angle = 0; angle < 360; angle++) {
        double rad = angle * 3.14159 / 180.0;
        int x = xc + (int)(r * cos(rad));
        int y = yc + (int)(r * sin(rad));
        plot(x, y);
    }
}

/* Draw Triangle */
void drawTriangle(int x1, int y1,
                  int x2, int y2,
                  int x3, int y3) {
    drawLine(x1, y1, x2, y2);
    drawLine(x2, y2, x3, y3);
    drawLine(x3, y3, x1, y1);
}

/* Redraw Entire Canvas */
void redrawCanvas() {
    int i;

    clearCanvas();

    for(i = 0; i < objectCount; i++) {

        switch(objects[i].type) {

        case LINE:
            drawLine(objects[i].p[0], objects[i].p[1],
                     objects[i].p[2], objects[i].p[3]);
            break;

        case RECTANGLE:
            drawRectangle(objects[i].p[0], objects[i].p[1],
                          objects[i].p[2], objects[i].p[3]);
            break;

        case CIRCLE:
            drawCircle(objects[i].p[0], objects[i].p[1],
                       objects[i].p[2]);
            break;

        case TRIANGLE:
            drawTriangle(objects[i].p[0], objects[i].p[1],
                         objects[i].p[2], objects[i].p[3],
                         objects[i].p[4], objects[i].p[5]);
            break;
        }
    }
}

/* Add Object */
void addObject(Shape s) {
    if(objectCount < MAX_OBJECTS) {
        objects[objectCount++] = s;
        redrawCanvas();
    }
}

/* Delete Object */
void deleteObject(int index) {
    int i;

    if(index < 0 || index >= objectCount)
        return;

    for(i = index; i < objectCount - 1; i++) {
        objects[i] = objects[i + 1];
    }

    objectCount--;
    redrawCanvas();
}

/* Modify Object */
void modifyObject(int index, Shape s) {
    if(index < 0 || index >= objectCount)
        return;

    objects[index] = s;
    redrawCanvas();
}

### Shape Parameters

Program:
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define WIDTH 80
#define HEIGHT 25
#define MAX_OBJECTS 100

char canvas[HEIGHT][WIDTH];
typedef enum { LINE = 1, RECT, CIRCLE, TRIANGLE } ShapeType;

typedef struct {
    ShapeType type;
    union {
        int coords[6]; // Flattened mapping: line/rect: [x1,y1,x2,y2], circle: [cx,cy,r], triangle: [x1,y1,x2,y2,x3,y3]
    } data;
} GraphicObject;

GraphicObject objects[MAX_OBJECTS];
int object_count = 0;

void clear_canvas() { memset(canvas, '_', sizeof(canvas)); }
void plot(int x, int y) { if (x >= 0 && x < WIDTH && y >= 0 && y < HEIGHT) canvas[y][x] = '*'; }

void draw_line(int x0, int y0, int x1, int y1) {
    int dx = abs(x1 - x0), sx = x0 < x1 ? 1 : -1;
    int dy = -abs(y1 - y0), sy = y0 < y1 ? 1 : -1;
    int err = dx + dy, e2;
    while (1) {
        plot(x0, y0);
        if (x0 == x1 && y0 == y1) break;
        e2 = 2 * err;
        if (e2 >= dy) { err += dy; x0 += sx; }
        if (e2 <= dx) { err += dx; y0 += sy; }
    }
}

void draw_rectangle(int x1, int y1, int x2, int y2) {
    int minx = x1 < x2 ? x1 : x2, maxx = x1 > x2 ? x1 : x2;
    int miny = y1 < y2 ? y1 : y2, maxy = y1 > y2 ? y1 : y2;
    for (int x = minx; x <= maxx; x++) { plot(x, miny); plot(x, maxy); }
    for (int y = miny; y <= maxy; y++) { plot(minx, y); plot(maxx, y); }
}

void draw_circle(int xc, int yc, int r) {
    int x = 0, y = r, d = 3 - 2 * r;
    while (y >= x) {
        int offsets[8][2] = {{x,y}, {-x,y}, {x,-y}, {-x,-y}, {y,x}, {-y,x}, {y,-x}, {-y,-x}};
        for (int i = 0; i < 8; i++) plot(xc + offsets[i][0], yc + offsets[i][1]);
        x++;
        if (d > 0) { y--; d += 4 * (x - y) + 10; }
        else d += 4 * x + 6;
    }
}

void render_all() {
    clear_canvas();
    for (int i = 0; i < object_count; i++) {
        int *d = objects[i].data.coords;
        switch (objects[i].type) {
            case LINE:     draw_line(d[0], d[1], d[2], d[3]); break;
            case RECT:     draw_rectangle(d[0], d[1], d[2], d[3]); break;
            case CIRCLE:   draw_circle(d[0], d[1], d[2]); break;
            case TRIANGLE: draw_line(d[0], d[1], d[2], d[3]);
                           draw_line(d[2], d[3], d[4], d[5]);
                           draw_line(d[4], d[5], d[0], d[1]); break;
        }
    }
}

void display_canvas() {
    for (int i = 0; i < 2; i++, putchar('\n')) for (int c = 0; c < WIDTH + 2; c++) putchar('-');
    for (int r = 0; r < HEIGHT; r++) {
        printf("|%.*s|\n", WIDTH, canvas[r]);
    }
    for (int c = 0; c < WIDTH + 2; c++) putchar('-'); putchar('\n');
}

int get_valid_int(const char* prompt, int min, int max) {
    int v, c;
    while (1) {
        printf("%s", prompt);
        if (scanf("%d", &v) == 1 && v >= min && v <= max) {
            while ((c = getchar()) != '\n' && c != EOF); return v;
        }
        printf("Invalid input.\n");
        while ((c = getchar()) != '\n' && c != EOF);
    }
}

void input_object_data(GraphicObject* o) {
    const char* prompts[][6] = {
        {"x1: ", "y1: ", "x2: ", "y2: "},
        {"corner1 x: ", "corner1 y: ", "corner2 x: ", "corner2 y: "},
        {"center x: ", "center y: ", "radius: "},
        {"v1 x: ", "v1 y: ", "v2 x: ", "v2 y: ", "v3 x: ", "v3 y: "}
    };
    int limits[] = {4, 4, 3, 6};
    int type_idx = o->type - 1;
    for (int i = 0; i < limits[type_idx]; i++) {
        int max_val = (o->type == CIRCLE && i == 2) ? 100 : ((i % 2 == 0) ? WIDTH - 1 : HEIGHT - 1);
        o->data.coords[i] = get_valid_int(prompts[type_idx][i], 0, max_val);
    }
}

int select_object(const char* action) {
    if (object_count == 0) { printf("No objects to %s.\n", action); return -1; }
    for (int i = 0; i < object_count; i++) printf("[%d] type %d\n", i, objects[i].type);
    return get_valid_int("Enter index: ", 0, object_count - 1);
}

void update_and_show() { render_all(); display_canvas(); }

int main() {
    clear_canvas();
    while (1) {
        printf("\n1.Add 2.Delete 3.Modify 4.Display 5.Clear 6.Exit\n");
        int ch = get_valid_int("Choice: ", 1, 6), idx;
        if (ch == 1 && object_count < MAX_OBJECTS) {
            objects[object_count].type = get_valid_int("1.Line 2.Rect 3.Circle 4.Triangle: ", 1, 4);
            input_object_data(&objects[object_count++]);
        } else if (ch == 1) puts("Limit reached.");
        else if (ch == 2 && (idx = select_object("delete")) >= 0) {
            for (int i = idx; i < object_count - 1; i++) objects[i] = objects[i + 1];
            object_count--;
        } else if (ch == 3 && (idx = select_object("modify")) >= 0) {
            input_object_data(&objects[idx]);
        } else if (ch == 5) object_count = 0;
        else if (ch == 6) return 0;
        update_and_show();
    }
}
